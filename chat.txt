в формате repo-gha-xxx или repo-gh-xxx, repo-git-xxx если более корректно

repo-gha-git-init:
---
#!/usr/bin/env bash
set -euo pipefail
{ set +x; } 2>/dev/null

if [ ! -e .git ]; then
  ( set -x; repo-gha-gen-wipe ) || exit 1
  ( set -x; git init -q -b main ) || exit 1
fi

( set -x; repo-gh-git-remote-set ) || exit 1

repo_full_name=$(repo-gh-full-name) || exit 1
if ! gh repo view "$repo_full_name" >/dev/null 2>&1; then
  ( set -x; repo-gh-create ) || exit 1
fi

has_local_head=0
git rev-parse --verify HEAD >/dev/null 2>&1 && has_local_head=1

has_local_files=0
if [ "$has_local_head" -eq 1 ] && [ -n "$(git ls-tree -r HEAD)" ]; then
  has_local_files=1
fi

remote_main_sha=$(git ls-remote origin refs/heads/main | awk '{print $1}')

if [ "$has_local_files" -eq 0 ]; then
  if [ -n "$remote_main_sha" ]; then
    ( set -x; git fetch -q origin main ) || exit 1
    ( set -x; git reset -q --soft origin/main ) || exit 1
  fi

  ( set -x; git checkout -q -B main ) || exit 1
  ( set -x; git add -A ) || exit 1

  if ! git diff --cached --quiet; then
    ( set -x; git commit --no-verify -q -m "chore: initial commit" ) || exit 1
  elif ! git rev-parse --verify HEAD >/dev/null 2>&1; then
    ( set -x; git commit --no-verify -q --allow-empty -m "chore: initial commit" ) || exit 1
  fi
fi

if git rev-parse --verify HEAD >/dev/null 2>&1; then
  ( set -x; git checkout -q -B main ) || exit 1
  local_head_sha=$(git rev-parse HEAD)
  if [ "$local_head_sha" != "$remote_main_sha" ]; then
    ( set -x; git push --no-verify -q -u origin main ) || exit 1
  fi
fi
